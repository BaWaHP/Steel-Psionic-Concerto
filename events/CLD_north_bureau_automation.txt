# Reset the repeatable subtree and request the next work direction.
country_event = {
	id = CLD_north_bureau_auto.3
	hidden = yes
	is_triggered_only = yes

	trigger = {
		tag = CLD
		NOT = { has_country_flag = CLD_north_bureau_mechanism_disabled }
	}

	immediate = {
		uncomplete_national_focus = {
			focus = CLD_north_bureau_meeting
			uncomplete_children = yes
			refund_political_power = no
		}
		clr_country_flag = CLD_250_a_flag
		clr_country_flag = CLD_250_b_flag
		clr_country_flag = CLD_250_c_flag
		clr_country_flag = CLD_250_d_flag
		clr_country_flag = CLD_north_bureau_automation_active
		set_country_flag = CLD_north_bureau_direction_event_pending
		country_event = CLD.250
	}
}
